\subsection{AILS Core Algorithm}

\begin{algorithm}[H]
\caption{AILS: Adaptive Incremental Line Search}
\label{alg:ails_complete}
\begin{algorithmic}[1]
\REQUIRE Grid $G=(V,E,w)$, start $s$, goal $g$, parameters $r_{\min}$, $r_{\max}$, $\alpha$, $\omega$, $\Delta r$
\ENSURE Path from $s$ to $g$, or $\emptyset$ if no $s$-$g$ path exists in~$G$

\STATE $I \gets$ \textsc{BuildIntegralImage}$(G)$ \COMMENT{$O(|V|)$ preprocessing}
\STATE $L \gets$ \textsc{BresenhamLine}$(s, g)$
\STATE $C_a \gets \emptyset$
\FORALL{$p \in L$}
    \STATE $\sigma(p) \gets$ \textsc{DensityQuery}$(I, p, \omega)$
    \STATE $r(p) \gets r_{\min} + \lfloor (r_{\max} - r_{\min}) \cdot \sigma(p)^\alpha \rfloor$
    \STATE $C_a \gets C_a \cup$ \textsc{ChebyshevBall}$(p, r(p))$
\ENDFOR

\REPEAT
    \STATE $path \gets$ \textsc{ConstrainedSearch}$(G, s, g, C_a)$
    \IF{$path \neq \emptyset$}
        \RETURN $path$
    \ENDIF
    \IF{$C_a = V$}
        \RETURN $\emptyset$ \COMMENT{Entire grid explored; no path exists}
    \ENDIF
    \STATE $C_a \gets C_a \cup$ \textsc{BFSExpand}$(C_a, \Delta r)$ \COMMENT{Fallback expansion}
\UNTIL{false}
\end{algorithmic}
\end{algorithm}

\subsection{Bresenham Corridor Generation}

\begin{algorithm}[H]
\caption{Bresenham Corridor Generation}
\label{alg:bresenham_corridor}
\begin{algorithmic}[1]
\REQUIRE Start point $s = (x_0, y_0)$, end point $e = (x_1, y_1)$, radius $r$
\ENSURE Set of corridor cells $C$

\STATE $C \gets \emptyset$
\STATE $dx \gets |x_1 - x_0|$
\STATE $dy \gets -|y_1 - y_0|$
\STATE $sx \gets$ \textbf{if} $x_0 < x_1$ \textbf{then} $1$ \textbf{else} $-1$
\STATE $sy \gets$ \textbf{if} $y_0 < y_1$ \textbf{then} $1$ \textbf{else} $-1$
\STATE $err \gets dx + dy$
\STATE $(x, y) \gets (x_0, y_0)$

\WHILE{$(x, y) \neq (x_1, y_1)$}
    \FORALL{$(dx', dy') \in$ \textsc{DiskOffsets}$(r)$}
        \STATE $C \gets C \cup \{(x + dx', y + dy')\}$
    \ENDFOR

    \STATE $e2 \gets 2 \cdot err$
    \IF{$e2 \geq dy$}
        \IF{$x = x_1$} \textbf{break} \ENDIF
        \STATE $err \gets err + dy$
        \STATE $x \gets x + sx$
    \ENDIF
    \IF{$e2 \leq dx$}
        \IF{$y = y_1$} \textbf{break} \ENDIF
        \STATE $err \gets err + dx$
        \STATE $y \gets y + sy$
    \ENDIF
\ENDWHILE

\RETURN $C$
\end{algorithmic}
\end{algorithm}

\subsection{Local Obstacle Density Estimation}

\begin{algorithm}[H]
\caption{Local Obstacle Density Estimation}
\label{alg:density}
\begin{algorithmic}[1]
\REQUIRE Grid $G$, point $p = (x, y)$, window half-size $\omega$
\ENSURE Normalised density $\sigma \in [0, 1]$

\STATE $obstacles \gets 0$
\STATE $total \gets 0$

\FOR{$dx = -\omega$ \TO $\omega$}
    \FOR{$dy = -\omega$ \TO $\omega$}
        \STATE $(nx, ny) \gets (x + dx, y + dy)$
        \IF{$(nx, ny)$ is within grid bounds}
            \STATE $total \gets total + 1$
            \IF{$G[nx][ny]$ is obstacle}
                \STATE $obstacles \gets obstacles + 1$
            \ENDIF
        \ENDIF
    \ENDFOR
\ENDFOR

\RETURN $obstacles / total$ \COMMENT{$O(\omega^2)$ per query, or $O(1)$ via integral image}
\end{algorithmic}
\end{algorithm}

\section{Theoretical Analysis}
% ============================================================================

\subsection{Proof of Completeness}

\begin{theorem}[AILS Completeness]
Let the base search algorithm~$\mathcal{A}$ be complete on finite graphs (e.g., A* with a consistent heuristic, Dijkstra, or BFS on unit-cost graphs).
If a path exists between start~$s$ and goal~$g$ in grid~$G$, AILS will find one after at most $\lceil D_{\max}/\Delta r \rceil$ expansion iterations, where $D_{\max}$ is the maximum Chebyshev distance from any cell on any $s$-$g$ path to the Bresenham line~$L$.
\end{theorem}

\begin{proof}
We prove completeness by induction on the expansion index~$k$.
Define the sequence of corridors $\{C_a^{(k)}\}_{k \geq 0}$ as:
\begin{align}
    C_a^{(0)} &= \{v \in V : \mathrm{occ}(v)=0,\; \exists\, p \in L \text{ with } \|v-p\|_{\infty} \leq r(p)\}, \\
    C_a^{(k+1)} &= C_a^{(k)} \cup \{v \in V \setminus C_a^{(k)} : \mathrm{occ}(v)=0,\; \exists\, u \in C_a^{(k)} \text{ with } \|v-u\|_{\infty} \leq \Delta r\}.
\end{align}

\textbf{Step 1 (Monotonicity).}
By construction, $C_a^{(k)} \subseteq C_a^{(k+1)}$ for all $k \geq 0$.
Moreover, if the constrained search on $C_a^{(k)}$ fails (no $s$-$g$ path within $C_a^{(k)}$), then there exists at least one traversable cell adjacent to the boundary of $C_a^{(k)}$ that is not in $C_a^{(k)}$ (otherwise $C_a^{(k)}$ would contain all traversable cells reachable from $s$, contradicting the existence of an $s$-$g$ path in $G$).
Hence $C_a^{(k)} \subsetneq C_a^{(k+1)}$ whenever expansion is triggered, i.e., $|C_a^{(k+1)}| > |C_a^{(k)}|$.

\textbf{Step 2 (Coverage growth).}
Let $V_{\mathrm{reach}} = \{v \in V : \mathrm{occ}(v)=0 \text{ and } v \text{ is reachable from } s \text{ via traversable cells}\}$.
We claim that for every $v \in V_{\mathrm{reach}}$ with $\min_{p \in L}\|v-p\|_{\infty} \leq r_{\max} + k \cdot \Delta r$, we have $v \in C_a^{(k)}$.

\emph{Base case ($k=0$):} Every traversable cell within Chebyshev distance $r(p) \leq r_{\max}$ of some reference-line cell $p$ is in $C_a^{(0)}$ by definition.

\emph{Inductive step:} Suppose the claim holds for $k$.
Any traversable cell $v$ with $\min_{p \in L}\|v-p\|_{\infty} \leq r_{\max} + (k{+}1)\Delta r$ is within Chebyshev distance $\Delta r$ of some cell $u$ satisfying $\min_{p \in L}\|u-p\|_{\infty} \leq r_{\max} + k \cdot \Delta r$.
By the inductive hypothesis, $u \in C_a^{(k)}$, so $v \in C_a^{(k+1)}$.

\textbf{Step 3 (Termination on finite grids).}
Since $|C_a^{(k)}|$ strictly increases at each expansion and $|V|$ is finite, there exists a finite $K \leq \lceil (\max(H, W_g) - r_{\min}) / \Delta r \rceil$ such that either (a) the constrained search finds a path at some iteration $k \leq K$, or (b) $C_a^{(K)} \supseteq V_{\mathrm{reach}}$.
In case (b), the search on $C_a^{(K)}$ is equivalent to an unconstrained search on the reachable subgraph of $G$.

\textbf{Step 4 (Completeness).}
Because $\mathcal{A}$ is complete on finite graphs by assumption, the search on $C_a^{(K)}$ finds a path whenever one exists in $G$.
Therefore, AILS is complete.

More precisely, if $\pi^{*}$ is any $s$-$g$ path in $G$ and $D_{\max} = \max_{v \in \pi^{*}} \min_{p \in L}\|v-p\|_{\infty}$, then setting $k^{*} = \lceil (D_{\max} - r_{\min}) / \Delta r \rceil$ guarantees $\pi^{*} \subseteq C_a^{(k^{*})}$, so AILS finds a path by iteration $k^{*}$ at the latest.

\textbf{Note.} The proof guarantees that AILS will find \emph{a} path; it does not guarantee that the path found before full expansion is globally optimal.
Global optimality holds only when the corridor covers at least one globally optimal path (see the main paper, Section~4).
\end{proof}

\subsection{Time Complexity Analysis}

\begin{theorem}[AILS Time Complexity]
Let the base algorithm~$\mathcal{A}$ run in $O(f(m))$ time on a graph with~$m$ vertices.
Without fallback, a single AILS invocation costs
$O(|V| + |L| \cdot r_{\max}^{2} + f(|C_a|))$.
With $k$ fallback rounds the total becomes
$O(|V| + |L| \cdot r_{\max}^{2} + \sum_{i=0}^{k} f(|C_a^{(i)}|))$.
In the worst case ($C_a$ grows to~$V$), the complexity matches that of an unconstrained search: $O(f(|V|))$.
\end{theorem}

\begin{proof}
\textbf{Integral image:} Built once in $O(|V|)$.

\textbf{Corridor generation:} The Bresenham line has $|L| = O(\max(H, W_g))$ cells.
With integral-image queries, the density at each cell costs $O(1)$; enumerating the Chebyshev ball of radius~$r(p)$ costs $O(r(p)^{2}) \leq O(r_{\max}^{2})$.
Total: $O(|L| \cdot r_{\max}^{2})$.

\textbf{Constrained search:} $\mathcal{A}$ runs on the subgraph induced by~$C_a$, costing $O(f(|C_a|))$.
For A* with a binary heap, $f(m) = O(m \log m)$.

\textbf{Fallback:} Each BFS boundary expansion visits $O(|C_a|)$ cells.
After $k$ rounds, the accumulated search cost is $\sum_{i=0}^{k} f(|C_a^{(i)}|)$.

\textbf{Worst case:} When $C_a$ grows to $V$, the cost is $O(f(|V|))$, matching unconstrained $\mathcal{A}$.
\end{proof}

\subsection{Space Complexity Analysis}

\begin{theorem}[AILS Space Complexity]
AILS uses $O(|V| + |C_a|)$ additional memory beyond the grid representation itself.
\end{theorem}

\begin{proof}
Space components:
\begin{itemize}
    \item Integral image: $O(|V|)$ (one integer per cell).
    \item Corridor hash set: $O(|C_a|)$.
    \item Base-algorithm data structures (open list, closed set, parent pointers): $O(|C_a|)$ because the search is restricted to~$C_a$.
\end{itemize}
Total additional memory: $O(|V| + |C_a|)$.
In the worst case ($C_a = V$), this is $O(|V|)$.
\end{proof}

\subsection{Python Implementation Excerpt}

\begin{lstlisting}[caption={Core AILS Corridor Builder}]
class CorridorBuilder:
    def __init__(self, grid, r_min=5, r_max=15, alpha=0.8, window=7):
        self.grid = grid
        self.r_min = r_min
        self.r_max = r_max
        self.alpha = alpha
        self.window = window

    def compute_local_density(self, x, y):
        """Compute obstacle density in local window."""
        obstacles = 0
        total = 0
        for dx in range(-self.window, self.window + 1):
            for dy in range(-self.window, self.window + 1):
                nx, ny = x + dx, y + dy
                if 0 <= nx < self.rows and 0 <= ny < self.cols:
                    total += 1
                    if self.grid[nx][ny] == 1:  # obstacle
                        obstacles += 1
        return obstacles / total if total > 0 else 0

    def adaptive_radius(self, density):
        """Compute adaptive corridor radius."""
        return self.r_min + int(
            (self.r_max - self.r_min) * (density ** self.alpha)
        )

    def build_corridor(self, start, goal):
        """Build adaptive corridor using Bresenham line."""
        corridor = set()
        line_points = self.bresenham_line(start, goal)

        for point in line_points:
            density = self.compute_local_density(*point)
            radius = self.adaptive_radius(density)

            # Add disk around each line point
            for dx in range(-radius, radius + 1):
                for dy in range(-radius, radius + 1):
                    if dx*dx + dy*dy <= radius*radius:
                        nx = point[0] + dx
                        ny = point[1] + dy
                        if 0 <= nx < self.rows and 0 <= ny < self.cols:
                            corridor.add((nx, ny))

        return corridor
\end{lstlisting}

\subsection{Comparison with Lee \& Lee 2025}

The experiments use the same benchmark suite as Lee \& Lee 2025 \cite{lee2025mcpp} to enable direct comparison. Key differences in experimental setup:

\begin{itemize}
    \item \textbf{Lee \& Lee 2025}: Multi-agent coverage path planning (MCPP) with $k \in \{2, 5, 10, 20, 30, 40\}$ agents
    \item \textbf{AILS (This work)}: Single-agent point-to-point pathfinding with corridor constraints
    \item \textbf{Shared benchmarks}: den312d, ht\_chantry, random-64-64-20 from Moving AI Lab
    \item \textbf{Scenarios}: Both use ``arbitrary'' and ``clutter'' configurations
\end{itemize}
